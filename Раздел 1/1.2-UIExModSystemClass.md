<h1 align="center">Класс UIExModSystem</h1>

Класс *UIExModSystem* является дочерним классом ModSystem. Его единственная задача – сократить реализацию класса ***ModSystem***, необходимого для создания пользовательского интерфейса, до минимума.


Как вы знаете, для того, чтобы начать писатель пользовательский интерфейс вы должны реализовать примерно следующую конструкцию:

---
Данная библиотека предполагает, что вы уже изучили [базовое руководство tModLoader по пользовательскому интерфейсу](https://github.com/tModLoader/tModLoader/wiki/Basic-UI-Element)

---

<details>
<summary>Базовая реализация ModSystem и UIState для написания UI</summary>

```csharp
public class ExampleMenuBar : UIState
{
    public override void OnInitialize() { }
}

[Autoload(Side = ModSide.Client)]
public class ExampleMenuBarSystem : ModSystem
{
    internal ExampleMenuBar ExmplMenuBar;
    private UserInterface _exampleMenuBar;
    public override void Load()
    {
        ExmplMenuBar = new ExampleMenuBar();
        ExmplMenuBar.Activate();
        _exampleMenuBar = new UserInterface();
    }
    public void SetState(bool state) => _exampleMenuBar.SetState(state ? ExmplMenuBar : null);
    public override void UpdateUI(GameTime gameTime) => _exampleMenuBar?.Update(gameTime);
    public override void ModifyInterfaceLayers(List<GameInterfaceLayer> layers)
    {
        int mouseTextIndex = layers.FindIndex(layer => layer.Name.Equals("Vanilla: Mouse Text"));
        if (mouseTextIndex != -1)
        {
            layers.Insert(mouseTextIndex, new LegacyGameInterfaceLayer(
                "YourMod: A Description",
                delegate
                {
                    _exampleMenuBar.Draw(Main.spriteBatch, new GameTime());
                    return true;
                },
                InterfaceScaleType.UI)
            );
        }
    }
}
```
</details>

Класс *UIExModSystem* дает вам возможность сократить данный код вплоть до одной строки:

```csharp
public class ExampleMenuBar : UIState
{
    public override void OnInitialize() { }
}

public class ExampleMenuBarSystem : UIExModSystem<ExampleMenuBar> { }
```

Вместо *ExampleMenuBar* может быть любой класс, наследуемый от *UIState*

<details>
<summary>Реализация класса UIExModSystem</summary>
```csharp
[Autoload(Side = ModSide.Client)]
public abstract class UIExModSystem<TUIState> : ModSystem 
    where TUIState : UIState, new()
{
    protected TUIState State { get; set; }
    protected UserInterface InterfaceState { get; set; }

    protected bool LastState { get; set; }

    protected virtual string ModInterfaceLayersName => "Vanilla: Mouse Text";
    protected virtual string ModInterfaceLayersDescription => "YourMod: A Description";

    /// <summary>
    ///     Позволяет задать состояние (отображается / скрыт) пользовательского интерфейса
    /// </summary>
    /// <param name="state">
    ///     Новое состояние (отображается / скрыт) пользовательского интерфейса
    /// </param>
    public virtual void SetState(bool state)
    {
        if (LastState == state)
            return;

        LastState = !LastState;
        InterfaceState.SetState(LastState ? State : null);
    }

    /// <summary>
    ///     Меняет состояние пользовательского интерфейса. Если отображается - скрывает; если скрыт - отображает.
    /// </summary>
    public virtual void SwitchState() => SetState(!LastState);

    /// <summary>
    ///     Отображает пользовательский интерфейс, если он скрыт.
    /// </summary>
    public virtual void Show() => SetState(true);

    /// <summary>
    ///     Скрывает пользовательский интерфейс, если он отображается.
    /// </summary>
    public virtual void Hide() => SetState(false);

    public override void Load()
    {
        State = new TUIState();
        State.Activate();
        InterfaceState = new UserInterface();
    }
    public override void UpdateUI(GameTime gameTime) => InterfaceState?.Update(gameTime);
    public override void ModifyInterfaceLayers(List<GameInterfaceLayer> layers)
    {
        int mouseTextIndex = layers.FindIndex(layer => layer.Name.Equals(ModInterfaceLayersName));
        if (mouseTextIndex != -1)
        {
            layers.Insert(mouseTextIndex, new LegacyGameInterfaceLayer(
                ModInterfaceLayersDescription,
                () =>
                {
                    InterfaceState.Draw(Main.spriteBatch, new GameTime());
                    return true;
                },
                InterfaceScaleType.UI)
            );
        }
    }
}
```

</details>
