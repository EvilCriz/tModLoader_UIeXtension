<h1 align="center">Класс <i>UIExStackPanel</i></h1>

`UIExStackPanel` - это контейнер компоновки, который располагает дочерние элементы в одну линию (стопку) по вертикали или по горизонтали.
`UIExStackPanel` определяет поле `public Styles.StyleLayoutContainerStackPanel StyleLayoutStackPanel { get; set; }`, которое хранит все стили компоновки для данного класса.

```csharp
public class StyleLayoutContainerStackPanel : IStyleLayoutContainer, IStyleCopy<StyleLayoutContainerStackPanel>
{
    public StyleDimension Spacing = StyleDimension.Empty;
    public bool Reverse = false;

    public void Set(StyleDimension spacing, bool reverse) { /* ... */ }
    public void Copy(StyleLayoutContainerStackPanel style) { /* ... */ }
    public StyleLayoutContainerStackPanel GetCopy() { /* ... */ }
}
```

- `Spacing`:
  - Отвечает за пространство между дочерними элементами `UIExStackPanel`.
  - Работает даже если `JustifyContent` и/или `AlignItems` == `UIExAlignment.Stretch`
- `Reverse`:
  - Флаг, отвечающий за расположение элементов в реверсивном порядке.
  - При `true` — все элементы будут располагаться в обратном порядке, сохраняя все прочие характеристики стиля.
<br>

Про `StyleLayoutChild`:
- Поле `JustifySelf` у дочерних элементов игнорируется.
- Поле `AlignSelf` переопределяет значение `AlignContent` для конкретного вложенного элемента.
- Поле `Margin` работает полноценно.

----

**НЕ РЕАЛИЗУЕТ** свой класс стиля для дочерних элементов.<br>

<h2 align="center">Пример использования <i>UIExStackPanel</i></h2>

<details>
<summary>Пример создания пользовательского интерфейса (без комментариев, разбор ниже)</summary>

```csharp
using Terraria.GameContent.UI.Elements;
using Terraria.UI;
using UIeXtenstion;
using UIeXtenstion.Enums;
using UIeXtenstion.Styles;

namespace UIeXtenstionExample;

public class ExampleMenuUI : UIExModSystem<ExamplePanelUI> { }

public class ExamplePanelUI : UIState
{
    // метод OnActivate() в данном случае используется только
    // для функции «горячей перезагрузки кода» в среде разработки.
    // для реального интерфейса правильнее было бы использовать
    // метод OnInitialize()
    public override void OnActivate()
    {
        RemoveAllChildren();

        UIPanel panel = new UIPanel();
        SetPrecentSizeCenter(0.8f, 0.8f, panel);

        UIExStackPanel HStackPanel = new();
        HStackPanel.Stretch();

        HStackPanel.StyleLayout.Set(
            orientation:        UIExOrientation.Horizontal,
            justifyContent:     UIExAlignment.Stretch,
            alignItems:         UIExAlignment.Stretch);
        HStackPanel.StyleLayoutStackPanel.Set(
            spacing:            new StyleDimension(0f, 0.05f),
            reverse:            false);

        panel.Append(HStackPanel);

        //////

        UIPanel HPanel1 = new(), HPanel2 = new(), HPanel3 = new(), HPanel4 = new();
        HStackPanel.Append(HPanel1);
        HStackPanel.Append(HPanel2);
        HStackPanel.Append(HPanel3);
        HStackPanel.Append(HPanel4);

        /////

        UIExStackPanel VStackPanel1 = new(), VStackPanel2 = new(), VStackPanel3 = new(), VStackPanel4 = new();
        SetPrecentSizeCenter(1f, 1f, VStackPanel1, VStackPanel2, VStackPanel3, VStackPanel4);
        HPanel1.Append(VStackPanel1);
        HPanel2.Append(VStackPanel2);
        HPanel3.Append(VStackPanel3);
        HPanel4.Append(VStackPanel4);

        VStackPanel1.StyleLayout.Set(
            orientation:        UIExOrientation.Vertical,
            justifyContent:     UIExAlignment.Start,
            alignItems:         UIExAlignment.Start);
        VStackPanel1.StyleLayoutStackPanel.Set(
            spacing:            new StyleDimension(0f, 0.1f),
            reverse:            false);

        UIElement selfStylePanel1 = AddCubs(VStackPanel1, pixelSize: 64f, count: 3)[1];
        VStackPanel1.GetOrRegisterStyleLayoutChild(selfStylePanel1)
            .Set(alignSelf: UIExAlignment.End);

        ////

        VStackPanel2.StyleLayout.Set(
            orientation:        UIExOrientation.Vertical,
            justifyContent:     UIExAlignment.Center,
            alignItems:         UIExAlignment.Center);
        VStackPanel2.StyleLayoutStackPanel.Set(
            spacing:            new StyleDimension(0f, 0.15f),
            reverse:            true);

        UIElement selfStylePanel2 = AddCubs(VStackPanel2, 16f, 32f, 48f, 64f)[2];

        VStackPanel2.GetOrRegisterStyleLayoutChild(selfStylePanel2).Set(
            justifySelf:        UIExAlignment.End, // ничего не делает, UIExStackPanel не учитывает JustifySelf,
            alignSelf:          UIExAlignment.Stretch,
            margin:             new UIExThickness(
                                    pixel:      false, 
                                    left:       0.1f, 
                                    top:        0.075f, 
                                    right:      0f, 
                                    bottom:     0.15f));

        ////

        VStackPanel3.StyleLayout.Set(
            orientation:        UIExOrientation.Vertical,
            justifyContent:     UIExAlignment.End,
            alignItems:         UIExAlignment.End);
        VStackPanel3.StyleLayoutStackPanel.Set(
            spacing:            new StyleDimension(0f, 0.20f),
            reverse:            false);

        AddCubs(VStackPanel3, 32f, 48f, 64f);

        ////

        VStackPanel4.StyleLayout.Set(
            orientation:        UIExOrientation.Vertical,
            justifyContent:     UIExAlignment.Stretch,
            alignItems:         UIExAlignment.Stretch);
        VStackPanel4.StyleLayoutStackPanel.Set(
            spacing:            new StyleDimension(0f, 0.20f),
            reverse:            false);

        AddCubs(VStackPanel4, pixelSize: 60f);
        AddCubs(VStackPanel4, pixelSize: 50f);
        AddCubs(VStackPanel4, pixelSize: 70f);

        ////

        Append(panel);
    }

    public void SetPrecentSizeCenter(float width, float height, params UIElement[] elements)
    {
        foreach (var element in elements)
        {
            element.Width.Set(0f, width);
            element.Height.Set(0f, height);
            element.HAlign = 0.5f;
            element.VAlign = 0.5f;
        }
    }

    public UIElement[] AddCubs(UIElement element, params float[] pixelSizes)
    {
        int count = pixelSizes.Length;
        var panels = new UIElement[count];
        for (int i = 0; i < count; i++)
            panels[i] = AddCubs(element, pixelSizes[i], count: 1)[0];
        return panels;
    }

    public UIElement[] AddCubs(UIElement element, float pixelSize, int count = 1)
    {
        var panels = new UIElement[count];
        for (int i = 0; i < count; i++)
        {
            panels[i] = new UIPanel();
            panels[i].Width.Set(pixelSize, 0f);
            panels[i].Height.Set(pixelSize, 0f);
            element.Append(panels[i]);
        }
        return panels;
    }
}
```

----

На изображении видно следующее:
1. Главная панель (размером в 80% экрана).
2. 4 вертикальные секции (размером с 100% высоты и равномерно растянутые по ширине, за исключением отступов).
3. В каждой из четырех секций: разное количество кубов с разным размером ребер.

----

Первое, что вы видите в коде:
```csharp
UIPanel panel = new UIPanel();
SetPrecentSizeCenter(0.8f, 0.8f, panel);
```
Это главная панель, которая будет добавлена в `UIState`. На ней будут располагаться `UIExStackPanel`.<br><br>
`SetPrecentSizeCenter` - это вспомогательный метод, которого нет в самой библиотеке, и который определен ниже в этом же классе.
Он получает N `UIElement`, устанавливаем им переданную высоту и ширину и центрирует. В данном случае наша `panel` имеет размер: 80%х80% экрана.<br><br>

----

Далее в коде идет первый `UIExStackPanel`:
```csharp
UIExStackPanel HStackPanel = new();
HStackPanel.Stretch();

HStackPanel.StyleLayout.Set(
    orientation:        UIExOrientation.Horizontal,
    justifyContent:     UIExAlignment.Stretch,
    alignItems:         UIExAlignment.Stretch);
HStackPanel.StyleLayoutStackPanel.Set(
    spacing:            new StyleDimension(0f, 0.05f),
    reverse:            false);

panel.Append(HStackPanel);

//////

UIPanel HPanel1 = new(), HPanel2 = new(), HPanel3 = new(), HPanel4 = new();
HStackPanel.Append(HPanel1);
HStackPanel.Append(HPanel2);
HStackPanel.Append(HPanel3);
HStackPanel.Append(HPanel4);
```

- `HStackPanel.Stretch();` - устанавливает размеры `HStackPanel`: 100% от родительского элемента.
- Ниже мы добавляем `HStackPanel` в главную панель `panel.Append(HStackPanel);`, что означает, что `HStackPanel` будет компоновать элементы по всей главной панели, т.е. по всему созданному интерфейсу.
- `Set` - это метод, который есть у любого класса стиля в библиотеке. Каждый Set стиля принимает значения для **ВСЕХ** существующих полей класса стиля.
- Его стили `StyleLayoutContainer StyleLayout`:
  - `Orientation`: Основная ось контейнера. `Vertical` - горизонтальная.
  - `JustifyContent` (выравнивание элементов по основной оси) (основная ось - горизонтальная): `stretch` - элементы занимают все доступное пространство по ширине.
  - `AlignItems` (выравнивание элементов по поперечной оси) (поперечной ось - вертикальная): `stretch` - элементы занимают все доступное пространство по высоте.
- Его стили `StyleLayoutContainerStackPanel StyleLayoutStackPanel`:
  - `Spacing`: 5% от размера основной оси контейнера (в данном случае: 5% от ширины `HStackPanel`).
  - `Reverse`: false - элементы располагаются в том порядке, в каком они будут добавлены в `HStackPanel`.


Ниже в `HStackPanel` добавлены 4 секции `HPanel_N`:
- `HStackPanel` сперва выделит место под `Spacing (5% * HStackPanel.Width * (HStackPanel.GetWithLayoutElementsCount() - 1))`
- Затем посчитает свободное пространство и поделит его между этими секциями.
- Результат этого вы можете посмотреть самостоятельно. На изображении это 4 самые вертикальные секции по всю высоту контейнера.

----

```csharp
UIExStackPanel VStackPanel1 = new(), VStackPanel2 = new(), VStackPanel3 = new(), VStackPanel4 = new();
SetPrecentSizeCenter(1f, 1f, VStackPanel1, VStackPanel2, VStackPanel3, VStackPanel4);
HPanel1.Append(VStackPanel1);
HPanel2.Append(VStackPanel2);
HPanel3.Append(VStackPanel3);
HPanel4.Append(VStackPanel4);
```

К каждой из созданных до этого секций (`HPanel_N`) создается свой `UIExStackPanel (VStackPanel_N)`.<br>
`VStackPanel_N` - будут компоновать кубы внутри каждой из секций.
Далее каждый из `VStackPanel_N` получает размеры 100% своего родителя и добавляются в соответствующие панель `HPanel_N`<br>

----

Хочу на этом месте подвести промежуточный итог.
На данный момент у нас система такая:
- `UIPanel panel` - главная панель. Имеет 80% размера экрана. Внутри нее расположен:<br>
  - `UIExStackPanel HStackPanel` - компоновщик внутри главной панели. Имеет 100% размера главной панели. Внутри него расположены:<br>
    - `UIPanel HPanel_1-4` - секции 1-4. Имеют 100% высоты `HStackPanel` и 25% от свободного пространства. Внутри них:<br>
      - `UIExStackPanel VStackPanel_1-4` - компоновщики внутри первой секции. Имеют 100% размера соответствующей секции. Внутри них<br>
          - N-количеством кубов в разным размером ребер.

Дальше для каждой `VStackPanel_1–4` задается стиль компоновки и добавляются другие `UIPanel` с помощью методов:
```csharp
public UIElement[] AddCubs(UIElement element, params float[] pixelSizes)
{
    int count = pixelSizes.Length;
    var panels = new UIElement[count];
    for (int i = 0; i < count; i++)
        panels[i] = AddCubs(element, pixelSizes[i], count: 1)[0];
    return panels;
}

public UIElement[] AddCubs(UIElement element, float pixelSize, int count = 1)
{
    var panels = new UIElement[count];
    for (int i = 0; i < count; i++)
    {
        panels[i] = new UIPanel();
        panels[i].Width.Set(pixelSize, 0f);
        panels[i].Height.Set(pixelSize, 0f);
        element.Append(panels[i]);
    }
    return panels;
}
```
Данные методы также созданы специально для данного примера и не существуют в самой библиотеке.<br><br>

Нет смысла разбирать весь код. Разберем только три части кода, где изменяются стили вложенных в контейнер элементов<br>

Первая часть:
```csharp
VStackPanel1.StyleLayout.Set(
    orientation:        UIExOrientation.Vertical,
    justifyContent:     UIExAlignment.Start,
    alignItems:         UIExAlignment.Start);
VStackPanel1.StyleLayoutStackPanel.Set(
    spacing:            new StyleDimension(0f, 0.1f),
    reverse:            false);

UIElement selfStylePanel1 = AddCubs(VStackPanel1, pixelSize: 64f, count: 3)[1];

VStackPanel1.GetOrRegisterStyleLayoutChild(selfStylePanel1)
    .Set(alignSelf: UIExAlignment.End);
```
Это этом коде задаются стили для контейнера в первой вертикальной панели (на изображении это самая левая вертикальная панель).
